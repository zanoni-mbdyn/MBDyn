        (\ty{real}) \bnt{Ex},

        (\ty{real}) \bnt{Dy},
        (\ty{real}) \bnt{Cy},
        (\ty{real}) \bnt{By},
        (\ty{real}) \bnt{Ey}
        # reference vx, damping coeff
        [\kw{low speed}, (\ty{real}) \bnt{vx_low}, (\ty{real}) \bnt{vx_dmp}];
